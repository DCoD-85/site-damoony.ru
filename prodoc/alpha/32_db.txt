================================================================
A. db.php — class db (фасад)
================================================================
СВОЙСТВА
  main, db_type = 'mysqli', driver (db_driver_interface),
  config (?db_config)

init()
  read_db_type()  — читает secure/db.secure → db_type
  load_files()    — include: driver, query, exception, config + <db_type>/{driver,query}
  create_driver() — new db_<type>($this), config = driver->config

CONVENIENCE
  db()             → driver
  open(localname='', reconnect=false) → driver
  get(localname='') → db_config_row|null
  config()         → db_config
  affected(), last_id(), escape()      — прокси к драйверу
  schema()         → new db_schema($this)

NEW API (обёртки над query)
  query($sql, $values=[], $types=[])   → db_query_interface
  select($sql, ...)                    — алиас query
  insert($table, $data, $types=[])     — INSERT с автозаголовками
  update($table, $data, $where, ...)   — UPDATE с автозаголовками
  delete($table, $where, ...)          — DELETE с автозаголовками

ЗАМЕЧАНИЯ
- db_type читается из secure/db.secure (JSON, ключ 'db_type').
- Fallback — 'mysqli'.
- load_files() подключает общий интерфейс + <db_type>/{driver,query}.

D. db/exception.php — class db_exception extends Exception
================================================================
__construct($message, $sql='', $values=[])
  Собирает: message + "\nSQL: " + SQL + "\nVALUES: " + JSON.

Используется в db_mysqli_query::raise() — при GL_DEBUG=true
ошибки кидаются как исключения, в prod — пишутся в error_log.

================================================================
E. db/config.php — db_config + db_config_row
================================================================
db_config
  Хранит rows[localname] => db_config_row + default (первая или
  помеченная default="true").

  path = root/secure/db.config    ← XML (не JSON!)
  init()  — simplexml_load_file, foreach <base>:
              пропускает base с disabled="true"
              читает <localname>, <dbname>, <dbhost>,
                    <username>, <password>, <comment>
              rows[localname] = row
              если default="true" → default = row
            Если default не задан — берётся первая строка.

  row($localname) : ?db_config_row
  default() : ?db_config_row
  all() : array
  has($localname) : bool

db_config_row
  localname, dbname, dbhost, username, password, comment
  is_valid()  — dbname && dbhost && username

ЗАМЕЧАНИЯ
- Конфиг — XML, в отличие от site.config (JSON).
- Ошибка парсинга → error() (die).

F. db/mysqli/driver.php — class db_mysqli implements db_driver_interface
================================================================
СВОЙСТВА
  db, main, config (new db_config), link (?mysqli),
  current (?db_config_row), last_error, last_affected, last_insert_id

open($localname='', $reconnect=false)
  1. row = config->default() | config->row($localname)
  2. Если уже подключены к тому же localname и не reconnect → return
  3. Если link !== null → close()
  4. mysqli_report(MYSQLI_REPORT_OFF)
  5. try { link = new mysqli(host, user, pass, db) } catch Throwable → error
  6. Если connect_errno → error
  7. set_charset('utf8mb4') + SET collation_connection

close()    — link->close()
link()     — raw mysqli (для внешнего использования)
status()   — 'connected: <localname>' | 'null'
error()    — link->error | last_error
affected() — link->affected_rows
last_id()  — link->insert_id
escape()   — link->real_escape_string
query()    — legacy: new db_mysqli_query($this)  (без SQL)

ЗАМЕЧАНИЯ
- PHP 8.3 — mysqli по умолчанию throws; здесь отключено.
- Один коннект на драйвер. Переключение баз — через open(localname).

================================================================
G. db/mysqli/query.php — class db_mysqli_query implements db_query_interface
================================================================
САМЫЙ БОЛЬШОЙ И ВАЖНЫЙ ФАЙЛ.

КОНВЕНЦИЯ SQL
  %s — строка
  %d — целое
  Парсятся в ? и подставляются через prepared statement.
  Пример: 'SELECT * FROM users WHERE id = %d AND lang = %s' + [5, 'ru']

ПУБЛИЧНЫЙ API
  query($sql, $values=[], $types=[])   — установить SQL
  one()                                 — LIMIT 1
  page($page, $size=null)               — LIMIT offset, size
  limit($n)                             — LIMIT n
  no_limit()                            — отключить авто-LIMIT
  result($assoc=true) : array
  row($assoc=true) : ?array
  scalar() : mixed
  count_rows() : int
  affected() : int
  insert_id() : int
  error() : string
  show() : string                       — html-превью SQL + values

ЛЕНИВОЕ ВЫПОЛНЕНИЕ
  execute() вызывается при первом result()/row()/scalar()/...
  Флаг $executed защищает от повторов.

АЛГОРИТМ execute()
  1. parse_sql(): %s/%d → ?, собрать типы ['s','d',...]
  2. apply_limit(): для SELECT дописать LIMIT (см. ниже)
  3. validate_types(): сверить кол-во placeholder'ов и values,
     привести типы (coerce)
  4. Если values пусто → mysqli::query() напрямую
  5. Иначе — mysqli::prepare() + bind_param + execute
  6. Результат — mysqli_result | affected_rows + insert_id

АВТО-LIMIT (apply_limit)
  Для SELECT:
    one()          → LIMIT 1
    limit(n)       → LIMIT n
    page(n, size)  → LIMIT (n-1)*size, size
                     (GET ?p=, ?p_max= переопределяют)
    иначе          → LIMIT DB_LIMIT_MAX (или 1000)

  Не применяется, если:
    no_limit() уже вызван
    в SQL уже есть ' LIMIT '
    query_type !== 'SELECT'

ТИПЫ И COERCION (validate_types)
  Если пользователь передал $types явно:
    нормализует '%s'/'%d' → 's'/'d', сверяет с SQL.
    Ошибка — mismatch count / mismatch type.
  Авто-коэрция:
    SQL %d + string → (int) в prod, exception в dev (GL_DEBUG=true)
    SQL %s + int    → (string)
    SQL %s + double → (string)

LEGACY API
  select($sql, $values, $args, $limit=false, $limitpage=0, $limitstep=10)
    — совместимость со старым интерфейсом
  insert($table, $names, $values, $types=[])
    — обёртка над insert_legacy
  insert_legacy(...) — тот же INSERT

DETECT_TYPE
  По первым 12 символам:
    SELECT/SHOW/DESCRIBE/DESC/EXPLAIN → 'SELECT'|'SHOW'
    INSERT/REPLACE                   → 'INSERT'
    UPDATE/DELETE                    → 'UPDATE'|'DELETE'
    CREATE/DROP/ALTER/TRUNCATE/RENAME → DDL
    иначе                            → 'OTHER'

  Влияет только на apply_limit (LIMIT только для SELECT).

БЕЗОПАСНОСТЬ
  Значения — только через bind_param (prepared).
  Имена таблиц/колонок в insert/update/delete фасада db.php — через
  backtick-экранирование, но БЕЗ параметризации.
  SQL для schema builder — через строки.

ЗАМЕЧАНИЯ
1. parse_sql() идёт посимвольно, не учитывает %s внутри строк 'aaa%sb'
   — редкий кейс, но потенциальный источник багов.
2. apply_limit для page() добавляет значения в $this->values и
   $this->types уже после parse_sql — порядок совпадает.
3. mysqli_stmt::get_result() требует mysqlnd (обычно есть).
4. affected_rows = -1 в mysqli после SELECT — здесь не критично,
   потому что result() и affected() разведены.
5. Для INSERT с явным списком колонок фасад db::insert() генерирует
   %d/%s автоматически, но не проверяет типы — они вычисляются по PHP-типу.

H. db/schema.php — db_schema + col()
================================================================
col(name, type='string', length=0, nullable=true, unsigned=false,
    auto=false, default=null, values=[]) : array
  Хелпер для описания колонки как массива.

db_schema
  Действия:
    create($table, $columns, $primary='id', $indexes=[], $foreign=[],
           $engine='InnoDB', $charset='utf8mb4', $collate='utf8mb4_unicode_ci',
           $if_not_exists=true)
    drop($table, $if_exists=true)
    truncate($table)
    rename($table, $new_name)

  Интроспекция:
    tables()         — SHOW TABLES
    exists($table)   — information_schema.tables
    columns_of($t)   — SHOW COLUMNS

  Построение:
    sql()  — итоговый SQL
    run()  — выполнить (db->select(sql)->affected())
    log()  — вернуть SQL

ТИПЫ (col_to_sql)
  int/integer/tinyint/smallint/mediumint/bigint → <TYPE>(length)
  string/varchar → VARCHAR(length|255)
  char → CHAR(length|1)
  text/mediumtext/longtext → <TYPE>
  enum → ENUM('a','b',...)
  bool/boolean → TINYINT(1)
  timestamp/datetime/date/time → <TYPE>
  decimal/float/double → <TYPE>(length)
  blob/mediumblob/longblob → <TYPE>
  json → JSON

  Модификаторы: UNSIGNED, AUTO_INCREMENT, NULL/NOT NULL, DEFAULT.

ЗАМЕЧАНИЯ
- CREATE TABLE всегда с ENGINE/CHARSET/COLLATE.
- Индексы: KEY или UNIQUE KEY через $idx['name'|'columns'|'unique'].
- Внешние ключи: FOREIGN KEY (col) REFERENCES table(ref).
- schema.php подгружается лениво — db::schema().

================================================================
СВОДНАЯ СХЕМА
================================================================
  main (db-модуль из sources/include.php)
    new db($main)
      read db_type (secure/db.secure)
      load drivers + config
      driver = new db_<type>($this)
        db_config → rows (secure/db.config, XML)
      config = driver->config

  $db = $main->db
  $db->open('visuo_damoony')            ← одна БД
  $db->query('SELECT ...', [...], [...]) → db_mysqli_query
    ->result() / ->row() / ->scalar()
    ->affected() / ->insert_id()

  $db->insert('files', ['slug'=>'a', ...]) — автогенерация INSERT
  $db->update('files', [...], ['id'=>1])
  $db->delete('files', ['id'=>1])

  $db->schema()
    ->create('files', [...], primary: 'id')
    ->run()
